cmake_minimum_required(VERSION 3.19)
project(AQCF LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "AQCF uses Linux sockets and process isolation. Use Linux or WSL2.")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/deps/install")
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(SEAL 4.1.2 CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd libzstd.so.1 REQUIRED)
find_library(GMP_LIBRARY NAMES gmp REQUIRED)
find_path(GMP_INCLUDE_DIR gmp.h REQUIRED)
set(AQCF_EIGEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/deps/_sources/eigen" CACHE PATH "Pinned Eigen headers")
if(NOT EXISTS "${AQCF_EIGEN_DIR}/Eigen/Core")
  message(FATAL_ERROR "Pinned Eigen headers missing: run python3 deps/fetch_dependencies.py")
endif()
add_subdirectory(deps)
add_subdirectory(native)
option(AQCF_BUILD_STRUCTURED "Build the structured CIFAR/Tiny ImageNet protocol" ON)
if(AQCF_BUILD_STRUCTURED AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/structured/CMakeLists.txt")
  add_subdirectory(structured)
endif()
enable_testing()
