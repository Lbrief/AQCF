# Same-function removals used by the structured-model study.
set(AQCF_ABLATION_SETTING "all_on" CACHE STRING "Structured study removal setting")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ablation_settings.json" AQCF_SETTINGS)
string(JSON SETTING_FLAGS ERROR_VARIABLE UNKNOWN_SETTING GET "${AQCF_SETTINGS}" "cifar_resnet18" "${AQCF_ABLATION_SETTING}")
if(UNKNOWN_SETTING)
  message(FATAL_ERROR "Unknown AQCF_ABLATION_SETTING: ${AQCF_ABLATION_SETTING}")
endif()
foreach(PROFILE cifar_resnet18 tiny_resnet18 cifar_mobilenetv2)
  string(JSON FLAGS ERROR_VARIABLE SETTING_ERROR GET "${AQCF_SETTINGS}" "${PROFILE}" "${AQCF_ABLATION_SETTING}")
  if(SETTING_ERROR)
    message(STATUS "${PROFILE}: ${AQCF_ABLATION_SETTING} does not apply; target omitted")
    continue()
  endif()
  set(TARGET "aqcf_${PROFILE}")
  add_executable(${TARGET} "${PROFILE}/inference_role.cpp")
  target_include_directories(${TARGET} BEFORE PRIVATE
    "${PROFILE}/include" "${PROFILE}/core/include" "${PROFILE}/generated")
  target_compile_features(${TARGET} PRIVATE cxx_std_17)
  target_compile_options(${TARGET} PRIVATE -O3 -maes -msse4.1 -mavx -mavx2 -faligned-new -Wno-deprecated-declarations)
  target_compile_definitions(${TARGET} PRIVATE NDEBUG EMP_USE_RANDOM_DEVICE=1
    AQCF_EXACT_QUARTER_BUILD=1 AQCF_QUARTER_PROFILE=2 AQCF_QUERY_FINALIZER_PUBLIC=0
    AQCF_ROOT_NATIVE_AUX_WIDE=1 AQCF_QUARTER_ADMITTED_REAL=1)
  if(PROFILE STREQUAL "cifar_mobilenetv2")
    target_compile_definitions(${TARGET} PRIVATE AQCF_BACKBONE_ADMITTED=1)
  else()
    target_compile_definitions(${TARGET} PRIVATE AQCF_RHOMBUS_ADMITTED=1)
  endif()
  string(JSON COUNT LENGTH "${FLAGS}")
  math(EXPR LAST "${COUNT}-1")
  foreach(INDEX RANGE 0 ${LAST})
    string(JSON FLAG MEMBER "${FLAGS}" ${INDEX})
    string(JSON VALUE GET "${FLAGS}" "${FLAG}")
    target_compile_definitions(${TARGET} PRIVATE "${FLAG}=${VALUE}")
  endforeach()
  target_link_libraries(${TARGET} PRIVATE aqcf_serial_sci aqcf_sci SEAL::seal)
  target_link_options(${TARGET} PRIVATE "-Wl,--wrap=bind")
endforeach()
